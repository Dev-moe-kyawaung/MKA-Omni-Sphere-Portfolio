Compose

Jetpack Compose Holograms

Compose Code Example


@Composable
fun HolographicCard(
    title: String,
    description: String,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier
            .background(
                brush = Brush.linearGradient(
                    colors = listOf(
                        Color(0x4000FFFF),
                        Color(0x40FF00FF)
                    )
                )
            )
            .clip(RoundedCornerShape(16.dp))
            .padding(24.dp)
    ) {
        Column {
            Text(
                text = title,
                style = MaterialTheme.typography.h6,
                color = Color.Cyan
            )
            Spacer(modifier = Modifier.height(8.dp))
            Text(
                text = description,
                style = MaterialTheme.typography.body1,
                color = Color.White
            )
        }
    }
}
